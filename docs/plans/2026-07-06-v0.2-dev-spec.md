```js
// Extract the first playable media file from a process argv array.
// dev 모드 argv: [electron.exe, '.', file] — '.'은 디렉토리라 걸러진다.
function extractFileArg(argv, cwd = process.cwd()) {
  for (const raw of argv.slice(1)) {
    if (!raw || raw.startsWith('-')) continue;
    const p = path.resolve(cwd, raw);
    let st;
    try { st = fs.statSync(p); } catch { continue; }
    if (!st.isFile()) continue;
    if (ALLOWED_INPUT_EXTS.has(path.extname(p).toLowerCase())) return p;
  }
  return null;
}
```

### 4) electron/main.js — 단일 인스턴스 락 + second-instance

```js

if (!gotLock) {
  app.quit();
} else {
  app.disableHardwareAcceleration();
  protocol.registerSchemesAsPrivileged([...기존...]);
  registerIpc();

      if (state.mainWindow.isMinimized()) state.mainWindow.restore();

  app.on('window-all-closed', ...기존...);
  app.on('activate', ...기존...);
```

`openFileInRenderer`는 `electron/windows.js`에 추가하고 main.js에서 import:

```js
// Route a file-open request to the renderer (so recent-files / playlist logic
// stays in one place). Queued until the renderer signals readiness.
function openFileInRenderer(p) {
  if (state.rendererReady && state.mainWindow && !state.mainWindow.isDestroyed()) {
    state.mainWindow.webContents.send('app:open-file', p);
  } else {
    state.pendingOpenPath = p;
  }
}
```

### 1) src/utils/resumeStore.ts (신규)

```ts
const KEY = 'offcut.player.positions';
const MAX_ENTRIES = 200;
const MIN_RESUME_SEC = 30;   // 이보다 앞이면 저장/복원 안 함
const TAIL_GUARD_SEC = 60;   // 끝에서 60초 이내면 "다 본 것"으로 간주해 삭제

export interface SavedPosition { position: number; duration: number; updatedAt: number }

type PositionMap = Record<string, SavedPosition>;

function normKey(p: string): string {
  return p.replace(/\//g, '\\').toLowerCase();
}

function load(): PositionMap { /* localStorage JSON.parse, 실패 시 {} */ }
function persist(map: PositionMap) { /* try-catch setItem */ }

export function savePosition(path: string, position: number, duration: number): void {
  if (!path || duration <= 0) return;
  const map = load();
  const k = normKey(path);
  if (position < MIN_RESUME_SEC || position > duration - TAIL_GUARD_SEC) {
    delete map[k]; // 앞부분이거나 끝까지 봤으면 기록 제거
  } else {
    map[k] = { position, duration, updatedAt: Date.now() };
  }
  // LRU 캡: MAX_ENTRIES 초과 시 updatedAt 오래된 것부터 제거
  const keys = Object.keys(map);
  if (keys.length > MAX_ENTRIES) {
    keys.sort((a, b) => map[a].updatedAt - map[b].updatedAt)
      .slice(0, keys.length - MAX_ENTRIES)
      .forEach((k2) => delete map[k2]);
  }
  persist(map);
}

export function lookupPosition(path: string): SavedPosition | null { /* load()[normKey] ?? null */ }
export function removePosition(path: string): void { /* delete + persist */ }
```

```tsx
      if (prev.file && prev.file !== s.filename) {
        savePosition(prev.file, prev.pos, prev.dur);
      }
      lastPosRef.current = { file: s.filename, pos: s.position, dur: s.duration };
      if (s.filename && !s.paused) savePosition(s.filename, s.position, s.duration);
    };
    const id = window.setInterval(tick, 5000);
    window.addEventListener('beforeunload', tick);
    return () => { window.clearInterval(id); window.removeEventListener('beforeunload', tick); };
  }, []);
```

   ```js
   const GIF_MAX_SEC = 30;
   ipcMain.handle('clip:gif', async (_evt, params) => {
     const ffmpegBin = resolveBinary('ffmpeg.exe');
     if (!ffmpegBin) throw new Error('ffmpeg.exe not found in resources/bin');
     const safeInput = validateMediaInput(params.input);
     const safeOutput = validateOutputPath(params.output);
     const start = Number(params.start); const end = Number(params.end);
     if (!Number.isFinite(start) || !Number.isFinite(end) || end <= start) throw new Error('잘못된 시작/끝 시각');
     if (end - start > GIF_MAX_SEC) throw new Error(`GIF는 최대 ${GIF_MAX_SEC}초까지 지원합니다`);
     const fps = Math.max(5, Math.min(30, Number(params.fps) || 15));
     const width = Math.max(120, Math.min(960, Number(params.width) || 480));
     const isWebp = /\.webp$/i.test(safeOutput);
     const args = [
       '-y', '-loglevel', 'error', ...FF_PROTOCOL_WHITELIST,
       '-ss', start.toFixed(3), '-to', end.toFixed(3), '-i', safeInput,
       ...(isWebp

   });
   ```
